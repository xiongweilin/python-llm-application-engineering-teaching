for attempt in range(1, max_attempts + 1):  # 依次产生 1 到上限的尝试编号，调用次数因此有边界
    try:  # 把一次操作包住，让不同失败类型进入不同分支

    pass  # 保持不变：pass 是合法语句，类名本身就是失败分类标签

class PermanentError(Exception):  # 给永久失败建立一个独立的可检查类型
    pass  # 保持不变：这里只建立分类，不增加额外行为

def run_with_retry(operation, max_attempts: int, sleep_fn):  # 接收操作、尝试上限和可替换的等待函数
    if max_attempts < 1:  # 没有至少一次尝试就无法执行，先在副作用前拒绝
        raise ValueError("max_attempts 必须至少为 1")  # 把调用者的参数错误明确报告出去

第 1/23 步 · 会话地图

本会话中文术语

术语按问题分组；每个词都对应具体代码责任。

暂时错误 temporary error
原因可能随时间消失，允许有限重试。
永久错误 permanent error
条件不变时再次尝试通常不会恢复。
指数退避 exponential backoff
让等待按 1、2、4 等逐次增长。
操作身份 operation identity
标识一次业务意图的稳定值。
原子预留 atomic reservation
不可被并发打断地取得唯一创建资格。
唯一约束 unique constraint
由数据库原子拒绝重复值。
行级规则 row-level rule
只依赖当前一行资料。
批次级规则 batch-level rule
依赖多行比较或整批汇总。
累计错误 error accumulation
继续执行独立规则并保存多个错误。
有界变化 bounded change
只改变一个明确条件，先观察结果，再说明稳定结构；开放式迁移留到后续阶段。
控制流 control flow
程序实际执行、跳转、返回和抛出的顺序。
接口约定 interface contract
调用者可依赖的输入、输出、异常和副作用规则。
